from fastapi import FastAPI, File, Form, UploadFile
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

@app.get("/health")
def health():
    return {"status": "ok"}

@app.post("/api/v1/analyze")
async def analyze(
    current_role: str = Form(...),
    experience_level: str = Form(...),
    current_skills: str = Form(...),
    target_role: str = Form(...),
    resume_file: UploadFile | None = File(None),
):
    return {
        "readinessScore": 72,
        "executiveSummary": "Two concise summary sentences...",
        "matchedSkills": ["Python", "SQL"],
        "partialSkills": ["Excel -> Modeling"],
        "missingSkills": ["PyTorch", "Docker"],
        "competencyScores": [
            {"area": "Technical", "current": 65, "target": 90},
            {"area": "Domain Expertise", "current": 70, "target": 85},
            {"area": "System Design", "current": 40, "target": 85},
            {"area": "Cloud & Ops", "current": 35, "target": 80},
            {"area": "Soft Skills", "current": 78, "target": 85}
        ],
        "roadmap": [{
            "skill": "PyTorch",
            "priority": "High",
            "estimatedTime": "3 weeks",
            "projectIdea": "Build an inference pipeline."
        }]
    }

class ChatRequest(BaseModel):
    message: str
    history: list[dict] = []
    context: dict = {}

@app.post("/api/v1/chat")
async def chat(req: ChatRequest):
    return {"reply": f"Coaching advice for: {req.message}"}

class ScrapeRequest(BaseModel):
    url: str

@app.post("/api/v1/scrape-job")
async def scrape_job(req: ScrapeRequest):
    return {"title": "Senior ML Engineer", "company": "Acme", "description": "Scraped job description..."}
